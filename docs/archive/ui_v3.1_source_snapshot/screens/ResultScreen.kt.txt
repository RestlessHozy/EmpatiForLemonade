@OptIn(ExperimentalLayoutApi::class, ExperimentalMaterial3Api::class)
@Composable
fun ModuleResultScreen(vm: EmpatiViewModel, modifier: Modifier = Modifier) {
    val presentation = vm.currentModulePresentation() ?: return
    val findings = if (presentation.isHypertension) {
        presentation.factorsToNote
    } else {
        presentation.identifiedCategories
    }
    var selectedTag by remember { mutableStateOf<String?>(null) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(MaterialTheme.colorScheme.background)
            .verticalScroll(rememberScrollState())
            .safeDrawingPadding()
            .padding(horizontal = 20.dp)
    ) {
        Spacer(Modifier.height(10.dp))
        Row(
            modifier = Modifier.fillMaxWidth(),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.SpaceBetween
        ) {
            BackIconButton(onClick = vm::onBack)
            SectionChip(
                text = if (presentation.isHypertension) "MODUL HIPERTENSI" else "MODUL KBG",
                containerColor = if (presentation.isHypertension)
                    MaterialTheme.colorScheme.primaryContainer
                else
                    MaterialTheme.colorScheme.secondaryContainer,
                contentColor = if (presentation.isHypertension)
                    MaterialTheme.colorScheme.onPrimaryContainer
                else
                    MaterialTheme.colorScheme.onSecondaryContainer
            )
        }

        Spacer(Modifier.height(18.dp))

        // Kartu Ringkasan Klinis Utama (Minimalis & Terpadu)
        TonalCard(
            containerColor = MaterialTheme.colorScheme.surface,
            borderColor = MaterialTheme.colorScheme.outlineVariant,
            shape = RoundedCornerShape(22.dp),
            contentPadding = PaddingValues(20.dp),
            modifier = Modifier.fillMaxWidth()
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                StatusPill(label = presentation.statusLabel, tone = presentation.tone)
                Box(
                    modifier = Modifier
                        .size(36.dp)
                        .background(
                            MaterialTheme.colorScheme.surfaceContainerLow,
                            RoundedCornerShape(10.dp)
                        ),
                    contentAlignment = Alignment.Center
                ) {
                    Icon(
                        imageVector = if (presentation.isHypertension)
                            Icons.Rounded.MonitorHeart
                        else
                            Icons.Rounded.Shield,
                        contentDescription = null,
                        tint = MaterialTheme.colorScheme.primary,
                        modifier = Modifier.size(18.dp)
                    )
                }
            }

            Spacer(Modifier.height(14.dp))
            Text(
                text = presentation.title,
                style = MaterialTheme.typography.headlineSmall,
                color = MaterialTheme.colorScheme.onSurface
            )
            Spacer(Modifier.height(6.dp))
            Text(
                text = presentation.conclusion,
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant
            )

            if (findings.isNotEmpty()) {
                Spacer(Modifier.height(16.dp))
                HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant)
                Spacer(Modifier.height(14.dp))
                Text(
                    text = if (presentation.isHypertension) "FAKTOR TERIDENTIFIKASI" else "INDIKASI TERIDENTIFIKASI",
                    style = MaterialTheme.typography.labelSmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
                Spacer(Modifier.height(10.dp))
                FlowRow(
                    horizontalArrangement = Arrangement.spacedBy(8.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    findings.forEach { item ->
                        ClinicalTagChip(
                            text = item,
                            onClick = { selectedTag = item }
                        )
                    }
                }
            }

            if (presentation.protectiveFactors.isNotEmpty()) {
                Spacer(Modifier.height(16.dp))
                HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant)
                Spacer(Modifier.height(14.dp))
                ProtectiveStrengthSection(items = presentation.protectiveFactors)
            }
        }

        if (presentation.supportNote != null) {
            Spacer(Modifier.height(12.dp))
            SupportStrip(note = presentation.supportNote)
        }

        Spacer(Modifier.height(12.dp))
        ActionPlanCard(
            title = "TINDAK LANJUT DISARANKAN",
            items = presentation.recommendations
        )

        if (!presentation.isComplete) {
            Spacer(Modifier.height(12.dp))
            CompactNotice(text = "Sebagian pertanyaan belum terjawab. Lengkapi untuk hasil optimal.")
        }

        Spacer(Modifier.height(22.dp))
        if (presentation.isHypertension) {
            PrimaryButton(text = "Lanjut ke Modul KBG", onClick = vm::proceedToGbv)
        } else {
            PrimaryButton(text = "Lihat Ringkasan Terintegrasi", onClick = vm::goToIntegrated)
        }

        Spacer(Modifier.height(14.dp))
        MinimalDisclaimer(text = presentation.disclaimer)
        Spacer(Modifier.height(20.dp))
    }

    selectedTag?.let { tag ->
        ClinicalDetailBottomSheet(
            tag = tag,
            onAskAi = {
                selectedTag = null
                vm.askAboutResult()
            },
            onDismiss = { selectedTag = null }
        )
    }
}

@OptIn(ExperimentalLayoutApi::class, ExperimentalMaterial3Api::class)
@Composable
fun IntegratedResultScreen(vm: EmpatiViewModel, modifier: Modifier = Modifier) {
    val presentation = vm.integratedPresentation() ?: return
    var selectedTag by remember { mutableStateOf<String?>(null) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(MaterialTheme.colorScheme.background)
            .verticalScroll(rememberScrollState())
            .safeDrawingPadding()
            .padding(horizontal = 20.dp)
    ) {
        Spacer(Modifier.height(10.dp))
        Row(
            modifier = Modifier.fillMaxWidth(),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.SpaceBetween
        ) {
            BackIconButton(onClick = vm::onBack)
            SectionChip(text = "RINGKASAN AKHIR")
        }

        Spacer(Modifier.height(18.dp))

        // 1. Kartu Klasifikasi Utama + Matriks Status 2 Modul
        TonalCard(
            containerColor = MaterialTheme.colorScheme.surface,
            borderColor = MaterialTheme.colorScheme.outlineVariant,
            shape = RoundedCornerShape(22.dp),
            contentPadding = PaddingValues(20.dp),
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(
                text = "KLASIFIKASI SKRINING",
                style = MaterialTheme.typography.labelSmall,
                color = MaterialTheme.colorScheme.primary
            )
            Spacer(Modifier.height(6.dp))
            Text(
                text = presentation.categoryLabel,
                style = MaterialTheme.typography.headlineSmall,
                color = MaterialTheme.colorScheme.onSurface
            )
            Spacer(Modifier.height(6.dp))
            Text(
                text = presentation.summary,
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant
            )

            Spacer(Modifier.height(16.dp))

            // Matriks Status 2 Kolom (Menggantikan 2 Kartu Paragraf Panjang)
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(10.dp)
            ) {
                ModuleStatusCell(
                    moduleCode = "HIPERTENSI",
                    statusLabel = presentation.hypertension.statusLabel,
                    tone = presentation.hypertension.tone,
                    modifier = Modifier.weight(1f)
                )
                ModuleStatusCell(
                    moduleCode = "KESEJAHTERAAN",
                    statusLabel = presentation.gbv.statusLabel,
                    tone = presentation.gbv.tone,
                    modifier = Modifier.weight(1f)
                )
            }

            Spacer(Modifier.height(12.dp))
            Surface(
                shape = RoundedCornerShape(12.dp),
                color = MaterialTheme.colorScheme.surfaceContainerLow,
                modifier = Modifier.fillMaxWidth()
            ) {
                Text(
                    text = presentation.relationshipExplanation,
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.padding(horizontal = 12.dp, vertical = 10.dp)
                )
            }
        }

        // 2. Faktor Kunci & Faktor Pelindung (Ringkas & Seimbang)
        if (presentation.factors.isNotEmpty() || presentation.protectiveFactors.isNotEmpty()) {
            Spacer(Modifier.height(12.dp))
            TonalCard(
                containerColor = MaterialTheme.colorScheme.surface,
                borderColor = MaterialTheme.colorScheme.outlineVariant,
                shape = RoundedCornerShape(20.dp),
                contentPadding = PaddingValues(18.dp),
                modifier = Modifier.fillMaxWidth()
            ) {
                if (presentation.factors.isNotEmpty()) {
                    Text(
                        text = "FAKTOR KUNCI",
                        style = MaterialTheme.typography.labelSmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                    Spacer(Modifier.height(10.dp))
                    FlowRow(
                        horizontalArrangement = Arrangement.spacedBy(8.dp),
                        verticalArrangement = Arrangement.spacedBy(8.dp)
                    ) {
                        presentation.factors.forEach { factor ->
                            ClinicalTagChip(
                                text = factor,
                                onClick = { selectedTag = factor }
                            )
                        }
                    }
                }

                if (presentation.factors.isNotEmpty() && presentation.protectiveFactors.isNotEmpty()) {
                    Spacer(Modifier.height(14.dp))
                    HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant)
                    Spacer(Modifier.height(14.dp))
                }

                if (presentation.protectiveFactors.isNotEmpty()) {
                    ProtectiveStrengthSection(items = presentation.protectiveFactors)
                }
            }
        }

        // 3. Tindak Lanjut Klinis
        Spacer(Modifier.height(12.dp))
        ActionPlanCard(
            title = "REKOMENDASI UTAMA",
            items = presentation.recommendations
        )

        Spacer(Modifier.height(22.dp))
        PrimaryButton(
            text = "Simpan ke Riwayat",
            onClick = vm::saveDualResult,
            trailingIcon = Icons.Rounded.BookmarkAdded
        )
        Spacer(Modifier.height(6.dp))
        GhostButton(
            text = "Tanyakan ke AI-EMPATI",
            onClick = vm::askAboutResult
        )

        Spacer(Modifier.height(12.dp))
        MinimalDisclaimer(text = presentation.disclaimer)
        Spacer(Modifier.height(20.dp))
    }

    selectedTag?.let { tag ->
        ClinicalDetailBottomSheet(
            tag = tag,
            onAskAi = {
                selectedTag = null
                vm.askAboutResult()
            },
            onDismiss = { selectedTag = null }
        )
    }
}

@Composable
private fun ModuleStatusCell(
    moduleCode: String,
    statusLabel: String,
    tone: PresentationTone,
    modifier: Modifier = Modifier
) {
    Surface(
        shape = RoundedCornerShape(14.dp),
        color = MaterialTheme.colorScheme.surfaceContainerLow,
        border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant),
        modifier = modifier
    ) {
        Column(modifier = Modifier.padding(12.dp)) {
            Text(
                text = moduleCode,
                style = MaterialTheme.typography.labelSmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant
            )
            Spacer(Modifier.height(8.dp))
            StatusPill(label = statusLabel, tone = tone)
        }
    }
}

            )
        }
    }
}

                        Text(
                            text = factor,
                            style = MaterialTheme.typography.bodySmall,
                            fontFamily = BodyFont,
                            fontWeight = FontWeight.Medium,
                            color = MaterialTheme.colorScheme.secondary
                        )
                    }
                }
            }
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
private fun ClinicalDetailBottomSheet(
    tag: String,
    onAskAi: () -> Unit,
    onDismiss: () -> Unit
) {
    val detail = remember(tag) { UserResultPresenter.explainClinicalTag(tag) }
    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)

    ModalBottomSheet(
        onDismissRequest = onDismiss,
        sheetState = sheetState,
        containerColor = MaterialTheme.colorScheme.surface,
        shape = RoundedCornerShape(topStart = 24.dp, topEnd = 24.dp)
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 24.dp)
                .padding(bottom = 28.dp)
        ) {
            Surface(
                shape = RoundedCornerShape(8.dp),
                color = MaterialTheme.colorScheme.surfaceContainerLow,
                border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant)
            ) {
                Text(
                    text = detail.categoryBadge,
                    style = MaterialTheme.typography.labelSmall,
                    fontFamily = MonoFont,
                    color = MaterialTheme.colorScheme.primary,
                    modifier = Modifier.padding(horizontal = 10.dp, vertical = 5.dp)
                )
            }

            Spacer(Modifier.height(12.dp))
            Text(
                text = detail.tag,
                style = MaterialTheme.typography.titleLarge,
                fontFamily = HeadingFont,
                color = MaterialTheme.colorScheme.onSurface
            )

            Spacer(Modifier.height(10.dp))
            Text(
                text = detail.explanation,
                style = MaterialTheme.typography.bodyMedium,
                fontFamily = BodyFont,
                color = MaterialTheme.colorScheme.onSurface
            )

            Spacer(Modifier.height(12.dp))
            Surface(
                shape = RoundedCornerShape(12.dp),
                color = MaterialTheme.colorScheme.surfaceContainerLow,
                border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant),
                modifier = Modifier.fillMaxWidth()
            ) {
                Text(
                    text = detail.linkageReminder,
                    style = MaterialTheme.typography.bodySmall,
                    fontFamily = BodyFont,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.padding(horizontal = 12.dp, vertical = 10.dp)
                )
            }

            Spacer(Modifier.height(20.dp))
            PrimaryButton(
                text = "Mengerti",
                onClick = onDismiss
            )
            Spacer(Modifier.height(8.dp))
            GhostButton(
                text = "Diskusikan di Tanya AI-EMPATI",
                onClick = onAskAi
            )
        }
    }
}

@Composable
private fun StatusPill(label: String, tone: PresentationTone) {
    val (container, content) = when (tone) {
        PresentationTone.POSITIVE -> Teal50 to Teal700
        PresentationTone.ATTENTION -> Amber50 to Amber700
        PresentationTone.INCOMPLETE -> MaterialTheme.colorScheme.surfaceVariant to MaterialTheme.colorScheme.onSurfaceVariant
        PresentationTone.NEUTRAL -> MaterialTheme.colorScheme.surfaceVariant to MaterialTheme.colorScheme.onSurfaceVariant
    }
    Surface(
        shape = CircleShape,
        color = container,
        border = BorderStroke(1.dp, content.copy(alpha = 0.2f))
    ) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            modifier = Modifier.padding(horizontal = 10.dp, vertical = 5.dp)
        ) {
            Box(
                Modifier
                    .size(6.dp)
                    .background(content, CircleShape)
            )
            Spacer(Modifier.width(6.dp))
            Text(
                text = label.uppercase(),
                color = content,
                fontFamily = MonoFont,
                fontSize = 10.sp,
                fontWeight = FontWeight.Bold,
                letterSpacing = 0.4.sp
            )
        }
    }
}
