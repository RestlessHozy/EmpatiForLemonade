private enum class ProfileField(val label: String, val icon: ImageVector) {
    NAME("Nama Lengkap", Icons.Rounded.Person),
    NIP("ID Akun (Email / No. HP / ID)", Icons.Rounded.Badge),
    INSTANSI("Domisili / Tempat Aktivitas", Icons.Rounded.Business),
    JABATAN("Pekerjaan / Peran Sehari-hari", Icons.Rounded.WorkOutline)
}

@Composable
fun ProfileScreen(vm: EmpatiViewModel, modifier: Modifier = Modifier) {
    var editingField by remember { mutableStateOf<ProfileField?>(null) }
    var editValue by remember { mutableStateOf("") }
    var showDeleteConfirm by remember { mutableStateOf(false) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(MaterialTheme.colorScheme.background)
            .verticalScroll(rememberScrollState())
            .statusBarsPadding()
            .padding(horizontal = 20.dp)
    ) {
        Spacer(Modifier.height(14.dp))
        Text(
            text = "Profil & Privasi",
            style = MaterialTheme.typography.headlineMedium,
            color = MaterialTheme.colorScheme.onBackground
        )
        Spacer(Modifier.height(18.dp))

        TonalCard(
            containerColor = MaterialTheme.colorScheme.surface,
            borderColor = MaterialTheme.colorScheme.outlineVariant,
            shape = RoundedCornerShape(24.dp),
            contentPadding = PaddingValues(20.dp),
            modifier = Modifier.fillMaxWidth()
        ) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                EmpatiAvatar(initials = vm.profile.initials(), size = 64.dp)
                Spacer(Modifier.width(16.dp))
                Column(modifier = Modifier.weight(1f)) {
                    SectionChip(text = "${vm.profile.golongan} · ${vm.profile.jenisAsn}")
                    Spacer(Modifier.height(8.dp))
                    Text(
                        text = vm.profile.name,
                        style = MaterialTheme.typography.headlineSmall,
                        color = MaterialTheme.colorScheme.onSurface
                    )
                    Spacer(Modifier.height(2.dp))
                    Text(
                        text = "${vm.profile.jabatan} · ${vm.profile.instansi}",
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }
            }
        }

        Spacer(Modifier.height(20.dp))
        Text(
            text = "Data Profil Pengguna",
            style = MaterialTheme.typography.titleSmall,
            color = MaterialTheme.colorScheme.onBackground,
            fontWeight = FontWeight.Bold
        )
        Spacer(Modifier.height(10.dp))

        TonalCard(
            containerColor = MaterialTheme.colorScheme.surface,
            borderColor = MaterialTheme.colorScheme.outlineVariant,
            shape = RoundedCornerShape(22.dp),
            contentPadding = PaddingValues(0.dp),
            modifier = Modifier.fillMaxWidth()
        ) {
            FieldRow(
                field = ProfileField.NAME,
                value = vm.profile.name,
                onEdit = {
                    editingField = ProfileField.NAME
                    editValue = vm.profile.name
                }
            )
            HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant)
            FieldRow(
                field = ProfileField.NIP,
                value = vm.profile.nip,
                // NIP adalah identitas login dan salt kredensial — tidak dapat diubah setelah registrasi.
                onEdit = null
            )
            HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant)
            FieldRow(
                field = ProfileField.INSTANSI,
                value = vm.profile.instansi,
                onEdit = {
                    editingField = ProfileField.INSTANSI
                    editValue = vm.profile.instansi
                }
            )
            HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant)
            FieldRow(
                field = ProfileField.JABATAN,
                value = vm.profile.jabatan,
                onEdit = {
                    editingField = ProfileField.JABATAN
                    editValue = vm.profile.jabatan
                }
            )
        }

        Spacer(Modifier.height(20.dp))
        Text(
            text = "Pengaturan & Keamanan",
            style = MaterialTheme.typography.titleSmall,
            color = MaterialTheme.colorScheme.onBackground,
            fontWeight = FontWeight.Bold
        )
        Spacer(Modifier.height(10.dp))

        TonalCard(
            containerColor = MaterialTheme.colorScheme.surface,
            borderColor = MaterialTheme.colorScheme.outlineVariant,
            shape = RoundedCornerShape(22.dp),
            contentPadding = PaddingValues(0.dp),
            modifier = Modifier.fillMaxWidth()
        ) {
            SettingRow(
                icon = Icons.Rounded.Lock,
                title = "Perlindungan Layar",
                subtitle = "Tangkapan layar diblokir aktif"
            )
            HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant)
            SettingRow(
                icon = Icons.Rounded.Notifications,
                title = "Pengingat Pemeriksaan",
                subtitle = "Nonaktif"
            )
            HorizontalDivider(color = MaterialTheme.colorScheme.outlineVariant)
            SettingRow(
                icon = Icons.Rounded.Info,
                title = "Tentang AI-EMPATI",
                subtitle = "Versi 3.0 · Inklusif Semua Kalangan"
            )
        }

        Spacer(Modifier.height(18.dp))

        GhostButton(
            text = "Keluar Sesi / Ganti Akun",
            onClick = vm::logoutAsn
        )

        Spacer(Modifier.height(10.dp))

        Surface(
            onClick = { showDeleteConfirm = true },
            shape = RoundedCornerShape(18.dp),
            color = MaterialTheme.colorScheme.errorContainer,
            border = BorderStroke(1.dp, MaterialTheme.colorScheme.error.copy(alpha = 0.25f)),
            modifier = Modifier.fillMaxWidth()
        ) {
            Row(
                modifier = Modifier.padding(horizontal = 18.dp, vertical = 15.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.Center
            ) {
                Icon(
                    imageVector = Icons.Rounded.DeleteOutline,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.onErrorContainer,
                    modifier = Modifier.size(20.dp)
                )
                Spacer(Modifier.width(10.dp))
                Text(
                    text = "Hapus Seluruh Data Saya dari Perangkat",
                    style = MaterialTheme.typography.labelLarge,
                    color = MaterialTheme.colorScheme.onErrorContainer,
                    fontWeight = FontWeight.Bold
                )
            }
        }

        Spacer(Modifier.height(28.dp))
    }

    editingField?.let { field ->
        AlertDialog(
            onDismissRequest = { editingField = null },
            shape = RoundedCornerShape(24.dp),
            containerColor = MaterialTheme.colorScheme.surface,
            title = {
                Text(
                    text = "Ubah ${field.label}",
                    style = MaterialTheme.typography.titleLarge
                )
            },
            text = {
                OutlinedTextField(
                    value = editValue,
                    onValueChange = { editValue = it },
                    singleLine = true,
                    shape = RoundedCornerShape(14.dp),
                    label = { Text(field.label) },
                    modifier = Modifier.fillMaxWidth()
                )
            },
            confirmButton = {
                TextButton(onClick = {
                    vm.updateProfile(applyEdit(vm.profile, field, editValue))
                    editingField = null
                }) {
                    Text("Simpan", fontWeight = FontWeight.Bold)
                }
            },
            dismissButton = {
                TextButton(onClick = { editingField = null }) {
                    Text("Batal")
                }
            }
        )
    }

    if (showDeleteConfirm) {
        AlertDialog(
            onDismissRequest = { showDeleteConfirm = false },
            shape = RoundedCornerShape(24.dp),
            containerColor = MaterialTheme.colorScheme.surface,
            title = {
                Text(
                    text = "Hapus Semua Data?",
                    style = MaterialTheme.typography.titleLarge
                )
            },
            text = {
                Text(
                    text = "Seluruh riwayat skrining, catatan evaluasi, dan profil Anda akan dihapus permanen dari perangkat ini.",
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
            },
            confirmButton = {
                TextButton(onClick = {
                    vm.clearAllData()
                    showDeleteConfirm = false
                }) {
                    Text(
                        text = "Hapus Permanen",
                        color = MaterialTheme.colorScheme.error,
                        fontWeight = FontWeight.Bold
                    )
                }
            },
            dismissButton = {
                TextButton(onClick = { showDeleteConfirm = false }) {
                    Text("Batal")
                }
            }
        )
    }
}

private fun applyEdit(profile: UserProfile, field: ProfileField, value: String): UserProfile = when (field) {
    ProfileField.NAME -> profile.copy(name = value)
    ProfileField.NIP -> profile // NIP immutable
    ProfileField.INSTANSI -> profile.copy(instansi = value)
    ProfileField.JABATAN -> profile.copy(jabatan = value)
}

@Composable
private fun FieldRow(
    field: ProfileField,
    value: String,
    onEdit: (() -> Unit)?
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .then(if (onEdit != null) Modifier.clickable(onClick = onEdit) else Modifier)
            .padding(horizontal = 18.dp, vertical = 14.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Box(
            modifier = Modifier
                .size(38.dp)
                .background(
                    MaterialTheme.colorScheme.surfaceContainerLow,
                    RoundedCornerShape(11.dp)
                ),
            contentAlignment = Alignment.Center
        ) {
            Icon(
                imageVector = field.icon,
                contentDescription = null,
                tint = MaterialTheme.colorScheme.primary,
                modifier = Modifier.size(18.dp)
            )
        }
        Spacer(Modifier.width(14.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = field.label,
                style = MaterialTheme.typography.labelSmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant
            )
            Spacer(Modifier.height(2.dp))
            Text(
                text = value,
                style = MaterialTheme.typography.bodyLarge,
                fontFamily = if (field == ProfileField.NIP) MonoFont else MaterialTheme.typography.bodyLarge.fontFamily,
                fontWeight = FontWeight.SemiBold,
                color = MaterialTheme.colorScheme.onSurface
            )
        }
        Icon(
            imageVector = if (onEdit != null) Icons.Rounded.Edit else Icons.Rounded.Lock,
            contentDescription = if (onEdit != null) "Ubah ${field.label}" else "${field.label} terkunci",
            tint = if (onEdit != null) MaterialTheme.colorScheme.primary
            else MaterialTheme.colorScheme.onSurfaceVariant,
            modifier = Modifier.size(18.dp)
        )
    }
}

@Composable
private fun SettingRow(
    icon: ImageVector,
    title: String,
    subtitle: String?
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 18.dp, vertical = 15.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Box(
            modifier = Modifier
                .size(38.dp)
                .background(
                    MaterialTheme.colorScheme.surfaceContainerLow,
                    RoundedCornerShape(11.dp)
                ),
            contentAlignment = Alignment.Center
        ) {
            Icon(
                imageVector = icon,
                contentDescription = null,
                tint = MaterialTheme.colorScheme.primary,
                modifier = Modifier.size(19.dp)
            )
        }
        Spacer(Modifier.width(14.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = title,
                style = MaterialTheme.typography.bodyLarge,
                fontWeight = FontWeight.SemiBold,
                color = MaterialTheme.colorScheme.onSurface
            )
            if (subtitle != null) {
                Spacer(Modifier.height(2.dp))
                Text(
                    text = subtitle,
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
            }
        }
    }
}
